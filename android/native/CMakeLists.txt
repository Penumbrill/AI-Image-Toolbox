cmake_minimum_required(VERSION 3.22)
project(aipt_diffusion LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "This JNI target must be built with the Android NDK")
endif()
set(AIPT_SD_SOURCE "" CACHE PATH "Verified stable-diffusion.cpp source checkout")
if(NOT EXISTS "${AIPT_SD_SOURCE}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set AIPT_SD_SOURCE to the verified offline source checkout")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(SD_CUDA OFF CACHE BOOL "" FORCE)
set(SD_VULKAN OFF CACHE BOOL "" FORCE)
set(SD_OPENCL OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    # Android x86_64 does not guarantee AVX/AVX2; keep this build emulator-safe.
    set(GGML_AVX OFF CACHE BOOL "" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
    set(GGML_FMA OFF CACHE BOOL "" FORCE)
    set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${AIPT_SD_SOURCE}" "${CMAKE_BINARY_DIR}/stable-diffusion")
add_library(aipt_diffusion SHARED aipt_diffusion.cpp)
target_link_libraries(aipt_diffusion PRIVATE stable-diffusion jnigraphics log)
target_compile_options(aipt_diffusion PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(aipt_diffusion PRIVATE "-Wl,-z,max-page-size=16384")
set_target_properties(aipt_diffusion PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/jni")
